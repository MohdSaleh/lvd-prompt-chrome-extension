# LVD Compiler - Chrome Extension

Real-time prompt enhancer for ChatGPT, Meta AI, Claude, Gemini.

**Matches your screenshot:** shows pills like `learn → adapt in context` and `adapt → Modify behavior...` right below the prompt input box, neat and easily accessible without complicating user flow.

## Features
- Works on: chat.openai.com, chatgpt.com, claude.ai, gemini.google.com, meta.ai
- Scans as you type using 738 LVD terms (longest-match-first)
- Shows yellow pills below input (like your screenshot)
- Click pill → replaces term in input with rephrase
- One-pass, debounced, non-intrusive

## Install (Developer Mode)
1. Go to chrome://extensions
2. Enable Developer mode
3. Click Load unpacked
4. Select this folder: lvd-chrome-extension

## How it works
- content.js finds prompt inputs (contenteditable divs and textareas)
- Builds regex from LVD_TERMS_SORTED (longest first)
- On input, finds unique matching terms (max 6 to keep neat)
- Creates #lvd-suggestion-bar with .lvd-pill buttons
- Clicking replaces that term via regex word-boundary replace

## Files
- manifest.json - MV3 manifest
- lvd-dictionary.js - 738 terms
- content.js - core injection logic
- styles.css - pill styling matching your screenshot
- popup.html - extension popup

## LVD Dictionary
738 terms covering:
- Core cognition: learn, understand, know, think, etc.
- Describe/Define tasks: Describe, Define, List, Outline, etc.
- Verification: double-check, be careful, don't miss anything
- Instructions 1-50: help me with this, give me an example, put this in a table
- Questions: What's going on here?, What changed?
- Verbs: Check, Fix, Improve, Analyze
- CAPS ops: ANCHOR, EVALUATE, VALIDATE, FINAL-AUDIT

// LVD Compiler - Content Script
// Supports ChatGPT, Claude, Gemini, Meta AI
// Shows pills like: [learn → adapt in context] below input, clickable to replace

(function () {
  const DEBUG = false;
  let currentInput = null;
  let suggestionBar = null;
  let lastText = '';
  let debounceTimer = null;

  function log(...args) { if (DEBUG) console.log('[LVD]', ...args); }

  function escapeRegExp(s) {
    return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  }

  function getRegexForTerms(terms) {
    const sorted = [...terms].sort((a, b) => b.length - a.length);
    const esc = sorted.map(escapeRegExp);
    // Use lookbehind if supported, fallback to \b
    try {
      return new RegExp('(?<!\\w)(' + esc.join('|') + ')(?!\\w)', 'gi');
    } catch (e) {
      return new RegExp('\\b(' + esc.join('|') + ')\\b', 'gi');
    }
  }

  function getDict() {
    if (typeof LVD_DICTIONARY !== 'undefined') return LVD_DICTIONARY;
    if (typeof window !== 'undefined' && window.LVD_DICTIONARY) return window.LVD_DICTIONARY;
    return {};
  }

  function getTerms() {
    if (typeof LVD_TERMS_SORTED !== 'undefined') return LVD_TERMS_SORTED;
    if (typeof window !== 'undefined' && window.LVD_TERMS_SORTED) return window.LVD_TERMS_SORTED;
    return Object.keys(getDict()).sort((a, b) => b.length - a.length);
  }

  let globalRegex = null;
  function buildRegex() {
    globalRegex = getRegexForTerms(getTerms());
  }
  buildRegex();

  function getTextFromInput(el) {
    if (!el) return '';
    if (el.tagName === 'TEXTAREA' || el.tagName === 'INPUT') {
      return el.value || '';
    }
    // contenteditable
    return el.innerText || el.textContent || '';
  }

  function setNativeValue(el, value) {
    // React & other frameworks track .value via a property descriptor; a direct
    // assignment is ignored and reverted on re-render. Use the native setter.
    const proto = el.tagName === 'TEXTAREA'
      ? (window.HTMLTextAreaElement && window.HTMLTextAreaElement.prototype)
      : el.tagName === 'INPUT'
        ? (window.HTMLInputElement && window.HTMLInputElement.prototype)
        : null;
    const desc = proto && Object.getOwnPropertyDescriptor(proto, 'value');
    if (desc && desc.set) desc.set.call(el, value);
    else el.value = value;
  }

  function placeCaretAtEnd(el) {
    try {
      const sel = window.getSelection();
      const range = document.createRange();
      range.selectNodeContents(el);
      range.collapse(false);
      sel.removeAllRanges();
      sel.addRange(range);
    } catch (e) { }
  }

  function replaceInNativeInput(el, term, rephrase, pos) {
    try { el.focus(); } catch (e) { }
    const val = el.value || '';
    let start = -1;
    let end = -1;

    if (typeof pos === 'number' && pos >= 0 && pos + term.length <= val.length) {
      if (val.slice(pos, pos + term.length).toLowerCase() === term.toLowerCase()) {
        start = pos;
        end = pos + term.length;
      }
    }

    if (start === -1) {
      const regex = getRegexForTerms([term]);
      regex.lastIndex = 0;
      const match = regex.exec(val);
      if (!match) return false;
      start = match.index;
      end = match.index + match[0].length;
    }

    const newVal = val.slice(0, start) + rephrase + val.slice(end);
    setNativeValue(el, newVal);
    el.dispatchEvent(new Event('input', { bubbles: true }));
    el.dispatchEvent(new Event('change', { bubbles: true }));
    try {
      el.selectionStart = el.selectionEnd = start + rephrase.length;
    } catch (e) { }
    return true;
  }

  function replaceInContentEditable(el, term, rephrase, pos) {
    try { el.focus(); } catch (e) { }
    const before = getTextFromInput(el);

    // Strip trailing browser placeholder newline if single line
    let cleanBefore = before;
    if (cleanBefore.endsWith('\n') && !cleanBefore.slice(0, -1).includes('\n')) {
      cleanBefore = cleanBefore.slice(0, -1);
    }

    let start = -1;
    let end = -1;
    if (typeof pos === 'number' && pos >= 0 && pos + term.length <= cleanBefore.length) {
      if (cleanBefore.slice(pos, pos + term.length).toLowerCase() === term.toLowerCase()) {
        start = pos;
        end = pos + term.length;
      }
    }

    let newText = '';
    if (start !== -1) {
      newText = cleanBefore.slice(0, start) + rephrase + cleanBefore.slice(end);
    } else {
      const termRegex = getRegexForTerms([term]);
      termRegex.lastIndex = 0;
      newText = cleanBefore.replace(termRegex, rephrase);
    }

    if (newText === cleanBefore) return false;

    // Select all content in the editor first so insertText replaces the entire prompt
    try {
      const sel = window.getSelection();
      const range = document.createRange();
      range.selectNodeContents(el);
      sel.removeAllRanges();
      sel.addRange(range);
    } catch (e) { }

    try {
      document.execCommand('selectAll', false, null);
    } catch (e) { }

    // Strategy 1: document.execCommand('insertText') replacing the selected text
    let success = false;
    try {
      success = document.execCommand('insertText', false, newText);
      if (success && getTextFromInput(el) !== before) {
        placeCaretAtEnd(el);
        return true;
      }
    } catch (e) { }

    // Strategy 2: Synthesized ClipboardEvent (paste)
    try {
      const dt = new DataTransfer();
      dt.setData('text/plain', newText);
      const pasteEv = new ClipboardEvent('paste', {
        bubbles: true,
        cancelable: true,
        clipboardData: dt
      });
      el.dispatchEvent(pasteEv);
      if (getTextFromInput(el) !== before) {
        placeCaretAtEnd(el);
        return true;
      }
    } catch (e) { }

    // Strategy 3: beforeinput event
    try {
      const bi = new InputEvent('beforeinput', {
        bubbles: true,
        cancelable: true,
        inputType: 'insertReplacementText',
        data: newText
      });
      el.dispatchEvent(bi);
      if (getTextFromInput(el) !== before) {
        placeCaretAtEnd(el);
        return true;
      }
    } catch (e) { }

    // Strategy 4: Direct innerText / textContent fallback
    try {
      el.innerText = newText;
      el.dispatchEvent(new InputEvent('input', { bubbles: true, data: newText, inputType: 'insertText' }));
      el.dispatchEvent(new Event('change', { bubbles: true }));
      placeCaretAtEnd(el);
      return true;
    } catch (e) { }

    return false;
  }

  // Generic fallback if needed
  function setTextInInput(el, newText) {
    if (!el) return false;
    try { el.focus(); } catch (e) { }

    if (el.tagName === 'TEXTAREA' || el.tagName === 'INPUT') {
      setNativeValue(el, newText);
      el.dispatchEvent(new Event('input', { bubbles: true }));
      el.dispatchEvent(new Event('change', { bubbles: true }));
      try { el.selectionStart = el.selectionEnd = newText.length; } catch (e) { }
      return true;
    }

    const before = getTextFromInput(el);
    try {
      const sel = window.getSelection();
      const range = document.createRange();
      range.selectNodeContents(el);
      sel.removeAllRanges();
      sel.addRange(range);
      const ok = document.execCommand('insertText', false, newText);
      if (ok && getTextFromInput(el) !== before) {
        placeCaretAtEnd(el);
        return true;
      }
    } catch (e) { }

    try {
      el.textContent = newText;
      el.dispatchEvent(new InputEvent('input', { bubbles: true, data: newText, inputType: 'insertText' }));
      placeCaretAtEnd(el);
      return true;
    } catch (e) { return false; }
  }

  function findPromptInputs() {
    const selectors = [
      '#prompt-textarea',
      'div#prompt-textarea[contenteditable="true"]',
      'div[data-lexical-editor="true"]',
      'div.ProseMirror[contenteditable="true"]',
      'div[contenteditable="true"][data-placeholder]',
      'div[contenteditable="true"][role="textbox"]',
      'div[contenteditable="true"]',
      'textarea[data-id="root"]',
      'textarea[placeholder*="Message"]',
      'textarea[placeholder*="Ask"]',
      'textarea',
      'div[aria-label*="Message"]',
      '[data-testid="chat-input"]'
    ];

    const candidates = [];
    for (const sel of selectors) {
      try {
        document.querySelectorAll(sel).forEach(el => {
          const rect = el.getBoundingClientRect();
          // Filter visible, reasonably sized, near bottom of viewport or focused
          if (rect.width > 100 && rect.height > 20 && rect.height < 400) {
            if (rect.bottom > window.innerHeight * 0.4 || document.activeElement === el) {
              candidates.push(el);
            }
          }
        });
      } catch (e) { }
    }

    // Prefer active element if it's an input
    const active = document.activeElement;
    if (active && (active.tagName === 'TEXTAREA' || active.getAttribute('contenteditable') === 'true' || active.tagName === 'INPUT')) {
      return [active, ...candidates.filter(c => c !== active)];
    }

    // Deduplicate and return bottom-most first
    const unique = [...new Set(candidates)];
    unique.sort((a, b) => b.getBoundingClientRect().bottom - a.getBoundingClientRect().bottom);
    return unique;
  }

  function createSuggestionBar() {
    if (document.getElementById('lvd-suggestion-bar')) return document.getElementById('lvd-suggestion-bar');
    const bar = document.createElement('div');
    bar.id = 'lvd-suggestion-bar';
    return bar;
  }

  function placeSuggestionBar(inputEl, bar) {
    if (!inputEl) return;

    // If bar already attached right before inputEl, keep it
    if (bar.parentElement && bar.nextElementSibling === inputEl) {
      return;
    }

    try {
      if (inputEl.parentElement) {
        // Inject directly inside the prompt box container, sitting right on top of the input field
        inputEl.parentElement.insertBefore(bar, inputEl);
        return;
      }
    } catch (e) { }

    try {
      if (inputEl.parentNode) {
        inputEl.parentNode.insertBefore(bar, inputEl);
        return;
      }
    } catch (e) { }

    try {
      document.body.appendChild(bar);
      const rect = inputEl.getBoundingClientRect();
      bar.style.position = 'fixed';
      bar.style.left = rect.left + 'px';
      bar.style.top = Math.max(0, rect.top - 46) + 'px';
      bar.style.width = rect.width + 'px';
      bar.style.zIndex = '99999';
    } catch (e) { }
  }

  function scanAndShow(inputEl) {
    const text = getTextFromInput(inputEl);
    if (!text || text.trim().length < 2) {
      hideBar();
      return;
    }
    if (text === lastText) return;
    lastText = text;

    const dict = getDict();
    const terms = getTerms();
    // Find all LVD matches in current text
    const matches = [];
    const seenKeys = new Set();
    let m;
    const regex = getRegexForTerms(terms);
    // Reset lastIndex
    regex.lastIndex = 0;

    while ((m = regex.exec(text)) !== null) {
      const raw = m[0];
      const key = raw.toLowerCase();
      if (seenKeys.has(key)) continue; // show unique terms only, like screenshot shows learn and adapt once each
      let dictKey = key;
      let rephrase = dict[key];
      if (!rephrase) {
        // Find case-insensitive
        const found = Object.keys(dict).find(k => k.toLowerCase() === key);
        if (found) {
          dictKey = found;
          rephrase = dict[found];
        }
      }
      if (rephrase) {
        matches.push({ term: raw, key: dictKey, rephrase: rephrase, pos: m.index });
        seenKeys.add(key);
      }
      if (matches.length >= 6) break; // limit to 6 pills to keep neat
      if (m[0].length === 0) regex.lastIndex++;
    }

    if (matches.length === 0) {
      hideBar();
      return;
    }

    showBar(inputEl, matches, text);
  }

  function showBar(inputEl, matches, originalText) {
    if (!suggestionBar) suggestionBar = createSuggestionBar();
    placeSuggestionBar(inputEl, suggestionBar);

    suggestionBar.innerHTML = '';
    matches.forEach(match => {
      const pill = document.createElement('button');
      pill.className = 'lvd-pill';
      pill.type = 'button';
      pill.innerHTML = `<span class="lvd-term">${escapeHtml(match.term)}</span><span class="lvd-arrow">→</span><span class="lvd-rephrase">${escapeHtml(match.rephrase)}</span>`;
      pill.title = `Replace "${match.term}" with "${match.rephrase}"`;
      pill.addEventListener('mousedown', (e) => {
        // Prevent button from stealing focus from the input box
        e.preventDefault();
      });
      pill.addEventListener('click', (e) => {
        e.preventDefault();
        e.stopPropagation();
        replaceTermInInput(inputEl, match.term, match.rephrase, match.pos);
      });
      suggestionBar.appendChild(pill);
    });

    suggestionBar.style.display = 'flex';
  }

  function hideBar() {
    if (suggestionBar) {
      suggestionBar.style.display = 'none';
      suggestionBar.innerHTML = '';
    }
  }

  function escapeHtml(s) {
    const div = document.createElement('div');
    div.textContent = s;
    return div.innerHTML;
  }

  function replaceTermInInput(inputEl, term, rephrase, pos) {
    if (!inputEl) inputEl = currentInput || document.activeElement;
    if (!inputEl) return;

    try {
      let changed = false;
      if (inputEl.tagName === 'TEXTAREA' || inputEl.tagName === 'INPUT') {
        changed = replaceInNativeInput(inputEl, term, rephrase, pos);
      } else {
        changed = replaceInContentEditable(inputEl, term, rephrase, pos);
      }

      if (changed) {
        log('Replaced', term, '->', rephrase);
      } else {
        // Fallback: try whole string replacement
        const text = getTextFromInput(inputEl);
        const termRegex = getRegexForTerms([term]);
        const newText = text.replace(termRegex, rephrase);
        if (newText !== text) {
          changed = setTextInInput(inputEl, newText);
        }
        if (!changed) console.warn('[LVD] replacement did not apply for', term);
      }

      setTimeout(() => {
        lastText = ''; // force rescan
        scanAndShow(inputEl);
      }, 100);
    } catch (e) {
      log('replace error', e);
    }
  }

  function attachToInput(inputEl) {
    if (!inputEl || inputEl.dataset.lvdAttached) return;
    inputEl.dataset.lvdAttached = 'true';
    currentInput = inputEl;
    log('Attached to', inputEl);

    const handler = () => {
      clearTimeout(debounceTimer);
      debounceTimer = setTimeout(() => scanAndShow(inputEl), 200);
    };

    inputEl.addEventListener('input', handler);
    inputEl.addEventListener('keyup', handler);
    inputEl.addEventListener('paste', () => setTimeout(handler, 50));
    inputEl.addEventListener('focus', handler);

    // Initial scan
    setTimeout(handler, 300);
  }

  function init() {
    // Initial find
    const inputs = findPromptInputs();
    inputs.slice(0, 2).forEach(attachToInput);

    // Watch for DOM changes (SPA navigation)
    const observer = new MutationObserver((mutations) => {
      const inputs = findPromptInputs();
      inputs.slice(0, 2).forEach(attachToInput);
    });

    observer.observe(document.body, { childList: true, subtree: true });

    // Also listen for focus changes
    document.addEventListener('focusin', (e) => {
      const target = e.target;
      if (target && (target.tagName === 'TEXTAREA' || target.getAttribute('contenteditable') === 'true' || target.tagName === 'INPUT')) {
        attachToInput(target);
        currentInput = target;
      }
    });

    // Periodic check for active input
    setInterval(() => {
      const active = document.activeElement;
      if (active && (active.tagName === 'TEXTAREA' || active.getAttribute('contenteditable') === 'true')) {
        if (!active.dataset.lvdAttached) attachToInput(active);
        if (currentInput !== active) {
          currentInput = active;
          scanAndShow(active);
        }
      }
    }, 1000);
  }

  // Start
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }

  // Also run after a delay for slow-loading apps
  setTimeout(init, 1500);
  setTimeout(init, 3000);
})();
